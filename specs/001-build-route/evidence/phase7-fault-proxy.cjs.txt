// Temporary manual verification stand. No application dependency or service endpoint.
const http=require('node:http');
const valid=['reorder','rate-limit','unavailable','malformed','mismatch','slow-create','lost-create'];
const mode=process.argv[process.argv.indexOf('--mode')+1];
if(!process.argv.includes('--mode')||!valid.includes(mode)){console.error('Usage: node route-builder-fault-proxy.cjs --mode '+valid.join('|'));process.exit(1);}
let sequence=0,incomingPosts=0,upstreamPosts=0,used=false;
const sockets=new Set(),upstreams=new Set(),timers=new Set();
function log(value){process.stdout.write(JSON.stringify({time:new Date().toISOString(),mode,incomingPosts,upstreamPosts,...value})+'\n');}
function delay(ms){return new Promise(resolve=>{const timer=setTimeout(()=>{timers.delete(timer);resolve();},ms);timers.add(timer);});}
const server=http.createServer(async(req,res)=>{
 const number=++sequence,method=req.method,pathname=new URL(req.url,'http://localhost').pathname;
 if(method==='POST')incomingPosts++;
 const entry={number,method,pathname};log({...entry,event:'received'});
 res.on('close',()=>{if(!res.writableEnded)log({...entry,event:'client_cancelled'});});
 try{
  let body=Buffer.alloc(0);for await(const chunk of req)body=Buffer.concat([body,chunk]);
  const get=method==='GET'&&pathname==='/v1/stops',post=method==='POST'&&pathname==='/v1/routes';
  if(!used&&get&&['rate-limit','unavailable','malformed'].includes(mode)){
   used=true;const code=mode==='rate-limit'?429:mode==='unavailable'?503:200;
   const payload=mode==='malformed'?'{':JSON.stringify({error:mode==='rate-limit'?'rate_limited':'upstream_unavailable'});
   res.writeHead(code,{'Content-Type':'application/json',...(code===429?{'Retry-After':'3'}:code===503?{'Retry-After':'2'}:{})});res.end(payload);log({...entry,event:'delivered',code,applied:true});return;
  }
  const applyPost=post&&!used&&['mismatch','lost-create'].includes(mode);if(applyPost)used=true;
  if(method==='POST')upstreamPosts++;
  const upstream=await new Promise((resolve,reject)=>{
   const request=http.request({hostname:'127.0.0.1',port:4000,path:req.url,method,headers:{...req.headers,host:'localhost:4000'}},response=>{
    const chunks=[];response.on('data',c=>chunks.push(c));response.on('end',()=>{upstreams.delete(request);resolve({code:response.statusCode,headers:{...response.headers},body:Buffer.concat(chunks)});});response.on('error',reject);
   });upstreams.add(request);request.on('error',error=>{upstreams.delete(request);reject(error);});request.end(body);
  });
  log({...entry,event:'upstream',code:upstream.code});
  if(applyPost&&mode==='mismatch'&&upstream.code===201){const data=JSON.parse(upstream.body);data.stops[0].instructions+='!';upstream.body=Buffer.from(JSON.stringify(data));delete upstream.headers['content-length'];delete upstream.headers['transfer-encoding'];upstream.headers['content-length']=String(upstream.body.length);}
  const q=new URL(req.url,'http://localhost').searchParams.get('q');
  let wait=0;if(get&&mode==='reorder'&&q==='North')wait=1500;
  if(post&&mode==='slow-create')wait=1500;
  if(applyPost&&mode==='lost-create'&&upstream.code===201)wait=12000;
  if(wait)await delay(wait);
  if(res.destroyed){log({...entry,event:'not_delivered',code:upstream.code,applied:!!wait||applyPost});return;}
  res.writeHead(upstream.code,upstream.headers);res.end(upstream.body);log({...entry,event:'delivered',code:upstream.code,applied:!!wait||applyPost});
 }catch(error){if(!res.destroyed){res.writeHead(502,{'Content-Type':'application/json'});res.end(JSON.stringify({error:'proxy_upstream_unavailable'}));}log({...entry,event:'proxy_error',code:502});}
});
server.on('connection',socket=>{sockets.add(socket);socket.on('close',()=>sockets.delete(socket));});
function shutdown(){for(const timer of timers)clearTimeout(timer);for(const request of upstreams)request.destroy();for(const socket of sockets)socket.destroy();server.close(()=>process.exit(0));}
process.on('SIGINT',shutdown);process.on('SIGTERM',shutdown);
server.listen(4001,'0.0.0.0',()=>log({event:'listening',port:4001}));
