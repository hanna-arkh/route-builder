const assert=require('node:assert/strict'),fs=require('node:fs'),{spawn}=require('node:child_process');
const port=4002,base='http://127.0.0.1:'+port,report={port,server:'unchanged server.js',groups:[]};
let child;
async function fresh(){
 if(child){child.kill('SIGTERM');await new Promise(resolve=>child.once('exit',resolve));}
 child=spawn(process.execPath,['server.js',String(port)],{cwd:process.cwd(),stdio:['ignore','pipe','pipe']});
 await new Promise((resolve,reject)=>{child.stdout.once('data',resolve);child.once('error',reject);child.once('exit',code=>reject(new Error('server exited '+code)));});
}
async function request(path,body,raw=false){const r=await fetch(base+path,body===undefined?undefined:{method:'POST',headers:{'Content-Type':'application/json'},body:raw?body:JSON.stringify(body)});return {status:r.status,retryAfter:r.headers.get('retry-after'),data:await r.json()};}
